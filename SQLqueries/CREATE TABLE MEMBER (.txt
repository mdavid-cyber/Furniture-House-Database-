CREATE TABLE BRANCH (
BranchID VARCHAR2(6) PRIMARY KEY,
Address VARCHAR2(100) NOT NULL,
Telephone VARCHAR2(12) NOT NULL,
Email VARCHAR2(50) NOT NULL
);

CREATE TABLE MEMBER (
MemberID VARCHAR2(8) PRIMARY KEY,
FirstName VARCHAR2(20) NOT NULL,
LastName VARCHAR2(20) NOT NULL,
EmailAddress VARCHAR2(50) NOT NULL,
PhoneNumber VARCHAR2(20) NOT NULL,
Address VARCHAR2(100) NOT NULL,
JoinDate DATE NOT NULL,
BranchID VARCHAR2(6) NOT NULL,
FOREIGN KEY (BranchID) REFERENCES Branch(BranchID)
);

CREATE TABLE Staff (
StaffID VARCHAR2(7) PRIMARY KEY,
FirstName VARCHAR2(20) NOT NULL,
LastName VARCHAR2(20) NOT NULL,
EmailAddress VARCHAR2(50) NOT NULL,
PhoneNumber VARCHAR2(20) NOT NULL,
JobPosition VARCHAR2(20) NOT NULL,
Salary Number(10) CONSTRAINT check_salary CHECK (Salary > 0),
BranchID VARCHAR2(6) NOT NULL,
ManagerID VARCHAR2(7),
FOREIGN KEY (ManagerID) REFERENCES Staff(StaffID),
FOREIGN KEY (BranchID) REFERENCES Branch(BranchID)
);

CREATE TABLE Furniture (
FurnitureID VARCHAR2(8) PRIMARY KEY,
StockNumber VARCHAR2(8) NOT NULL,
ItemTitle VARCHAR2(50) NOT NULL,
Colour VARCHAR2(8),
ItemSize VARCHAR2(10) CONSTRAINT check_itemsize CHECK (ItemSize IN ('KingSize', 'Small', 'Medium', 'Large')),
Price NUMBER(8) CHECK (Price > 0),
BranchID VARCHAR2(6) NOT NULL,
FOREIGN KEY (BranchID) REFERENCES Branch(BranchID)
);

CREATE TABLE Rental (
RentalID VARCHAR2(7) PRIMARY KEY,
DateOut DATE NOT NULL,
DateIn DATE NOT NULL,
TotalCost NUMBER(20) CONSTRAINT check_totalcost CHECK (TotalCost > 0),
MemberID VARCHAR2(8) NOT NULL,
FurnitureID VARCHAR2(8) NOT NULL,
FOREIGN KEY (MemberID) REFERENCES Member(MemberID),
FOREIGN KEY (FurnitureID) REFERENCES Furniture(FurnitureID)
);
COMMIT;
